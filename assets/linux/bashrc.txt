  if [[ -z "$ORIG" ]]; then
              ORIG=$PS1
                fi
                  TITLE="\[\e]2;$*\a\]"
                    PS1=${ORIG}${TITLE}
 }

# Fastread
fastread() {
    local wpm="$1"
    local file="$2"
    local skip=0   # change here if you want different default skip

    if [[ -z "$wpm" || -z "$file" ]]; then
        echo "Usage: fastread <wpm> <file>"
        return 1
    fi

    if [[ ! -f "$file" ]]; then
        echo "File not found: $file"
        return 1
    fi

    # Calculate delay from WPM
    local delay
    delay=$(awk "BEGIN {print 60/$wpm}")

    # Hide cursor for clean effect
    tput civis

    tail -n +"$((skip+1))" "$file" |
    tr -s '[:space:]' '\n' |
    while read -r word; do
        rows=$(tput lines)
        cols=$(tput cols)

                row=$((rows / 2))
        col=$(((cols - ${#word}) / 2))

        clear
        tput cup "$row" "$col"
        printf "\033[7m%s\033[0m" "$word"

        sleep "$delay"
    done

    # Restore cursor
    tput cnorm
}


# Show highlighted keywords; Logic: words between *; like *imp word*
showbold() {
find . -type f -name "*.txt" -print0 |
while IFS= read -r -d '' file; do
    matches=$(grep -oP '\*\S(?:.*?\S)?\*' "$file" \
              | sed 's/^\*//; s/\*$//' \
              | paste -sd ", " -)
    if [ -n "$matches" ]; then
        echo "${file#./}: $matches"
    fi
done
}

# Show recent modified files in terminal
recent() {
        find . -type f -iname "$1" -printf '%T@ %p\0' | sort -zk1nr | head -z -n $2 | cut -z -f2- -d' ' | tr '\0' '\n'
}
